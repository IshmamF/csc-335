;; Abstract syntax trees for environments

(define scheme-value?
  (lambda (v) #t))

(define-datatype environment environment?
  (empty-env-record)
  (extended-env-record
    (syms (list-of symbol?))
    (vals (list-of scheme-value?))
    (env environment?)))

;;corresponding implementation of the environment data type

(define list-find-position (lambda (sym los)
                             (list-index (lambda (sym1)(eqv? sym1 sym)) los)))

(define list-index (lambda (pred ls)
                     (cond ((null? ls) #f)
                           ((pred (car ls)) 0)
                           (else (let ((list-index-r (list-index pred (cdr ls))))
                                   (if (number? list-index-r)
                                       (+ list-index-r 1)
                                       #f))))))

(define empty-env
  (lambda () (empty-env-record)))

(define extend-env
  (lambda (syms vals env)
    (extended-env-record syms vals env)))

(define apply-env
  (lambda (env sym)
    (cases environment env
      (empty-env-record () (eopl:error 'apply-env "No binding for ~s" sym))
      (extended-env-record (syms vals env)
                           (let ((pos (list-find-position sym syms)))
                             (if (number? pos)
                                 (list-ref vals pos)   
                                 (apply-env env sym)))))))




;; with all of this in place, we are asked to implement environment-to-list, as illustrated
;; in the text (in the current context, this is perhaps useful as a means of
;; exploring the behavior of the interpreter

(define environment-to-list
  (lambda (env)
    (cases environment env
      (empty-env-record () (list 'empty-env-record))
      (extended-env-record (syms vals env)
                           (list 'extended-env-record syms vals (environment-to-list env))))))

  (closure (ids (list-of symbol?))
           (body expression?)
           (env environment?)))

(define apply-procval
  (lambda (proc args)
    (cases procval proc
      (closure (ids body env)
        (eval-expression body (extend-env ids args env))))))

;; here is the interpreter from Figure 3.2, modified for Exercises 3.16, 3.17 and 3.19

(define eval-program
  (lambda (pgm)
    (cases program pgm
      (a-program (body)
        (eval-expression body (init-env))))))


(define eval-expression
  (lambda (exp env)
    
    (letrec ((cond-aux (lambda (test-exps conseq-exps)
                         (cond ((null? test-exps) 0)
                               ((true-value? (eval-expression (car test-exps) env))
                                (eval-expression (car conseq-exps) env))
                               (else (cond-aux (cdr test-exps) (cdr conseq-exps)))))))
      ;; preserves usual normal order aspect of cond evaluation

      
      (cases expression exp
        (lit-exp (datum) datum)
        (var-exp (id) (apply-env env id))
        (if-exp (test-exp true-exp false-exp)
                (if (true-value? (eval-expression test-exp env))
                    (eval-expression true-exp env)
                    (eval-expression false-exp env)))
        
        (cond-exp (test-exps conseq-exps)
                  (cond-aux test-exps conseq-exps))
        
        (let-exp (ids rands body)
                 (let ((args (eval-rands rands env)))
                   (eval-expression body (extend-env ids args env))))
        
        (primapp-exp (prim rands)
                     (let ((args (eval-rands rands env)))
                       (apply-primitive prim args)))
        
        (proc-exp (ids body)
                  (closure ids body env))
        
        (app-exp (rator rands)
                 (let ((proc (eval-expression rator env))
                       (args (eval-rands rands env)))
                   (if (procval? proc)
                       (apply-procval proc args)
                       (eopl:error 'eval-expression
                                   "Attempt to apply non-procedure ~s" proc))))


        ))))

(define true-value?
  (lambda (x)
    (not (zero? x))))

(define eval-rands
  (lambda (rands env)
    (map (lambda (x) (eval-rand x env)) rands)))

(define eval-rand
  (lambda (rand env)
    (eval-expression rand env)))


(define apply-primitive
  (lambda (prim args)
    (letrec ((fold-right (lambda (op init seq)
                           (cond ((null? seq) init)
                                 (else (op (car seq) (fold-right op init (cdr seq)))))))
             
             (fold-left (lambda (op init seq)
                          (letrec ((iter (lambda (result rest)
                                           (if (null? rest)
                                               result
                                               (iter (op result (car rest))
                                                     (cdr rest))))))
                            (iter init seq))))
             
             (chain-op (lambda (compare)
                  (let ((last #f))
                    (lambda (result curr)
                      (if (not last)
                          (set! last curr)
                          (cond ((not result) 0) ;; 0 instead of #f due to the
                                ;; requirements of these problems
                                ((compare last curr) 1) ;; 1 instead of #t ...
                                (else 0)))))))
      )

    
    (cases primitive prim
      (add-prim () (fold-right + 0 args))
      (subtract-prim () (fold-right - 0 args))
      (mult-prim () (fold-right * 1 args))
      (incr-prim () (+ (car args) 1))
      (decr-prim () (- (car args) 1))
      (minus-prim () (* (car args) -1))
      (cons-prim () (cons (car args) (cadr args)))
      (car-prim () (caar args))
      (cdr-prim () (cdar args))
      (list-prim () (fold-right cons '() args))
      
      (eq-prim () (if (eq? (car args) (cadr args)) 1 0))
      
      (equal-prim () (fold-left (chain-op equal?) #t args))
      (zero-prim () (if (zero? (car args)) 1 0))
      
      (greater-prim () (fold-left (chain-op >) #t args))
      (less-prim () (fold-left (chain-op <) #t args))
      
      (null-prim () (if (null? (car args)) 1 0))
      )
    
    )))



(define init-env
  (lambda ()
    (extend-env
     '(i v x emptylist)
     '(1 5 10 ())
     (empty-env))))
